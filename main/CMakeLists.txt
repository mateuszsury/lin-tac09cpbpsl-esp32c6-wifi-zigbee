if(NOT DEFINED KLIMA_HA_TRANSPORT)
    set(KLIMA_HA_TRANSPORT "MQTT")
endif()
string(TOUPPER "${KLIMA_HA_TRANSPORT}" KLIMA_HA_TRANSPORT_NORMALIZED)

set(KLIMA_SRCS
    "main.c"
    "ac_protocol.c"
    "bridge_service.c"
    "capture_store.c"
    "telemetry.c"
    "wifi_service.c"
    "web_service.c"
    "nts0104_preflight.c"
)
set(KLIMA_REQUIRES
    app_update json esp_driver_gpio esp_driver_rmt esp_driver_uart esp_coex
    esp_event esp_http_server esp_netif esp_partition esp_timer esp_wifi mdns
    nvs_flash
)

if(KLIMA_HA_TRANSPORT_NORMALIZED STREQUAL "MQTT")
    list(APPEND KLIMA_SRCS "mqtt_service.c")
    list(APPEND KLIMA_REQUIRES mqtt)
elseif(NOT KLIMA_HA_TRANSPORT_NORMALIZED STREQUAL "ZIGBEE" AND
       NOT KLIMA_HA_TRANSPORT_NORMALIZED STREQUAL "NONE")
    message(FATAL_ERROR "KLIMA_HA_TRANSPORT must be MQTT, ZIGBEE, or NONE")
endif()

idf_component_register(SRCS ${KLIMA_SRCS} INCLUDE_DIRS "." REQUIRES ${KLIMA_REQUIRES})

if(KLIMA_HA_TRANSPORT_NORMALIZED STREQUAL "MQTT")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_HA_MQTT_ACTIVE=1 KLIMA_HA_ZIGBEE_ACTIVE=0
        KLIMA_WIFI_FALLBACK_AP_ENABLED=1)
elseif(KLIMA_HA_TRANSPORT_NORMALIZED STREQUAL "ZIGBEE")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_HA_MQTT_ACTIVE=0 KLIMA_HA_ZIGBEE_ACTIVE=1
        KLIMA_WIFI_FALLBACK_AP_ENABLED=0)
else()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_HA_MQTT_ACTIVE=0 KLIMA_HA_ZIGBEE_ACTIVE=0
        KLIMA_WIFI_FALLBACK_AP_ENABLED=1)
endif()

if(NOT DEFINED KLIMA_BRIDGE_MODE)
    set(KLIMA_BRIDGE_MODE "SNIFFER")
endif()
string(TOUPPER "${KLIMA_BRIDGE_MODE}" KLIMA_BRIDGE_MODE_NORMALIZED)

# Final KAmodNTS0104PW wiring shared by every public profile:
# main TX -> B1/A1 -> GPIO4; GPIO6 -> A2/B2 -> panel RX;
# panel TX -> B3/A3 -> GPIO7; GPIO0 -> A4/B4 -> main RX; OE -> GPIO3.
set(KLIMA_FINAL_GPIO_DEFINITIONS
    KLIMA_MAIN_RX_GPIO_NUM=4 KLIMA_PANEL_TX_GPIO_NUM=6
    KLIMA_PANEL_RX_GPIO_NUM=7 KLIMA_MAIN_TX_GPIO_NUM=0)

if(KLIMA_BRIDGE_MODE_NORMALIZED STREQUAL "SNIFFER" OR
   KLIMA_BRIDGE_MODE_NORMALIZED STREQUAL "PASSIVE")
    # Safe default: RX-only; OE and every output remain disabled.
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_BRIDGE_ACTIVE=0 KLIMA_BRIDGE_MIRROR_PROBE=0
        KLIMA_HEADLESS_PROBE=0 KLIMA_MAIN_EDGE_FORWARD=0
        KLIMA_MAIN_HARDWARE_FORWARD=0 KLIMA_CANONICAL_SNIFFER=1
        KLIMA_NTS0104_PREFLIGHT=1 KLIMA_NTS0104_OUTPUTS_ENABLED=0
        ${KLIMA_FINAL_GPIO_DEFINITIONS})
elseif(KLIMA_BRIDGE_MODE_NORMALIZED STREQUAL "PANEL_DIAG")
    # Detached display diagnosis: translator may be enabled, but no UART TX.
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_BRIDGE_ACTIVE=0 KLIMA_BRIDGE_MIRROR_PROBE=0
        KLIMA_HEADLESS_PROBE=0 KLIMA_MAIN_EDGE_FORWARD=0
        KLIMA_MAIN_HARDWARE_FORWARD=0 KLIMA_CANONICAL_PANEL_DIAG=1
        KLIMA_NTS0104_PREFLIGHT=1 KLIMA_NTS0104_OUTPUTS_ENABLED=1
        ${KLIMA_FINAL_GPIO_DEFINITIONS})
elseif(KLIMA_BRIDGE_MODE_NORMALIZED STREQUAL "PANEL_BENCH")
    # Detached display only. Never use while the real main PCB is attached.
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_BRIDGE_ACTIVE=0 KLIMA_BRIDGE_MIRROR_PROBE=0
        KLIMA_HEADLESS_PROBE=0 KLIMA_MAIN_EDGE_FORWARD=0
        KLIMA_MAIN_HARDWARE_FORWARD=0 KLIMA_CANONICAL_PANEL_BENCH=1
        KLIMA_PANEL_BENCH_EMULATOR=1 KLIMA_NTS0104_PREFLIGHT=1
        KLIMA_NTS0104_OUTPUTS_ENABLED=1 ${KLIMA_FINAL_GPIO_DEFINITIONS})
elseif(KLIMA_BRIDGE_MODE_NORMALIZED STREQUAL "BRIDGE")
    # Transparent forwarding with external actuation deliberately disabled.
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_BRIDGE_ACTIVE=1 KLIMA_HYBRID_BRIDGE=1
        KLIMA_CONTROL_VERIFIED=0 KLIMA_PANEL_HARDWARE_FORWARD=0
        KLIMA_BRIDGE_MIRROR_PROBE=0 KLIMA_HEADLESS_PROBE=0
        KLIMA_MAIN_EDGE_FORWARD=0 KLIMA_MAIN_HARDWARE_FORWARD=0
        KLIMA_CANONICAL_BRIDGE=1 KLIMA_NTS0104_PREFLIGHT=1
        KLIMA_NTS0104_OUTPUTS_ENABLED=1 ${KLIMA_FINAL_GPIO_DEFINITIONS})
elseif(KLIMA_BRIDGE_MODE_NORMALIZED STREQUAL "MITM_NTS")
    # Active, response-confirmed control. Never selected implicitly.
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        KLIMA_BRIDGE_ACTIVE=1 KLIMA_HYBRID_BRIDGE=1
        KLIMA_CONTROL_VERIFIED=1 KLIMA_SYNCED_TX_PROBE=0
        KLIMA_PANEL_HARDWARE_FORWARD=1 KLIMA_PANEL_LOOP_CONTROL=1
        KLIMA_BOOT_FORCE_OFF_DIAG=0 KLIMA_A4_B4_DIAGNOSTIC=0
        KLIMA_MAIN_TX_OPEN_DRAIN=0 KLIMA_BRIDGE_MIRROR_PROBE=0
        KLIMA_HEADLESS_PROBE=0 KLIMA_MAIN_EDGE_FORWARD=0
        KLIMA_MAIN_HARDWARE_FORWARD=0 KLIMA_CANONICAL_MITM_NTS=1
        KLIMA_NTS0104_PREFLIGHT=1 KLIMA_NTS0104_OUTPUTS_ENABLED=1
        ${KLIMA_FINAL_GPIO_DEFINITIONS})
else()
    message(FATAL_ERROR
        "KLIMA_BRIDGE_MODE must be SNIFFER, PASSIVE, PANEL_DIAG, PANEL_BENCH, BRIDGE, or MITM_NTS")
endif()

message(STATUS "Klima bridge build profile: ${KLIMA_BRIDGE_MODE_NORMALIZED}")
message(STATUS "Klima Home Assistant transport: ${KLIMA_HA_TRANSPORT_NORMALIZED}")
